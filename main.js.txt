const prefersReducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
const isFinePointer = window.matchMedia("(hover: hover) and (pointer: fine)").matches;

function initNavigation(){
  const header=document.getElementById("site-header");
  const toggle=document.querySelector(".menu-toggle");
  const links=document.getElementById("nav-links");
  const navAnchors=[...document.querySelectorAll("[data-nav]")];
  const sections=[...document.querySelectorAll("section[id]")];
  const updateHeader=()=>header?.classList.toggle("is-scrolled",window.scrollY>20);
  updateHeader();window.addEventListener("scroll",updateHeader,{passive:true});
  toggle?.addEventListener("click",()=>{const open=toggle.getAttribute("aria-expanded")==="true";toggle.setAttribute("aria-expanded",String(!open));links?.classList.toggle("is-open",!open)});
  links?.addEventListener("click",e=>{if(e.target.closest("a")){links.classList.remove("is-open");toggle?.setAttribute("aria-expanded","false")}});
  if("IntersectionObserver" in window){
    const observer=new IntersectionObserver(entries=>{entries.forEach(entry=>{if(entry.isIntersecting){const id=entry.target.id;navAnchors.forEach(a=>a.classList.toggle("is-active",a.getAttribute("href")===`#${id}`))}})},{rootMargin:"-35% 0px -55% 0px",threshold:0});
    sections.forEach(s=>observer.observe(s));
  }
}

function initRevealAnimations(){
  const els=[...document.querySelectorAll("[data-reveal]")];
  if(prefersReducedMotion||!("IntersectionObserver" in window)){els.forEach(el=>el.classList.add("is-visible"));return}
  const observer=new IntersectionObserver(entries=>{entries.forEach(entry=>{if(entry.isIntersecting){entry.target.classList.add("is-visible");observer.unobserve(entry.target)}})},{threshold:.12,rootMargin:"0px 0px -7% 0px"});
  els.forEach((el,i)=>{el.style.transitionDelay=`${Math.min(i%4,3)*55}ms`;observer.observe(el)});
}

function initScrollProgress(){
  const fill=document.getElementById("scroll-fill");
  const activeTrace=document.getElementById("timeline-active");
  if(activeTrace){const len=activeTrace.getTotalLength();activeTrace.style.strokeDasharray=String(len);activeTrace.style.strokeDashoffset=String(len);activeTrace.dataset.length=String(len)}
  let ticking=false;
  const update=()=>{const max=document.documentElement.scrollHeight-window.innerHeight;const progress=max>0?Math.min(1,Math.max(0,window.scrollY/max)):0;if(fill)fill.style.height=`${progress*100}%`;if(activeTrace){const rect=document.getElementById("experience")?.getBoundingClientRect();const vh=window.innerHeight;if(rect){const local=Math.min(1,Math.max(0,(vh*.72-rect.top)/(rect.height+vh*.3)));const len=Number(activeTrace.dataset.length||1000);activeTrace.style.strokeDashoffset=String(len*(1-local))}}ticking=false};
  window.addEventListener("scroll",()=>{if(!ticking){requestAnimationFrame(update);ticking=true}},{passive:true});update();
}

function initProjectFilters(){
  const filters=[...document.querySelectorAll("[data-filter]")];
  const projects=[...document.querySelectorAll(".project-item")];
  filters.forEach(button=>button.addEventListener("click",()=>{const value=button.dataset.filter;filters.forEach(b=>b.classList.toggle("is-active",b===button));projects.forEach(card=>{const categories=(card.dataset.categories||"").split(/\s+/);const show=value==="all"||categories.includes(value);card.classList.toggle("is-hidden",!show);if(show&&!prefersReducedMotion)card.animate([{opacity:0,transform:"translateY(12px)"},{opacity:1,transform:"translateY(0)"}],{duration:350,easing:"cubic-bezier(.22,1,.36,1)"})})}));
}

function initPCBAnimation(){
  const visual=document.getElementById("pcb-visual");if(!visual)return;
  if(prefersReducedMotion){visual.querySelectorAll("animateMotion").forEach(el=>el.remove())}
  const xOut=document.getElementById("coord-x"),yOut=document.getElementById("coord-y");
  if(isFinePointer&&!prefersReducedMotion){visual.addEventListener("pointermove",e=>{const r=visual.getBoundingClientRect();const x=(e.clientX-r.left)/r.width-.5,y=(e.clientY-r.top)/r.height-.5;visual.style.transform=`perspective(1000px) rotateY(${x*3}deg) rotateX(${-y*3}deg)`;if(xOut)xOut.textContent=String(Math.round((x+.5)*999)).padStart(3,"0");if(yOut)yOut.textContent=String(Math.round((y+.5)*999)).padStart(3,"0")});visual.addEventListener("pointerleave",()=>visual.style.transform="")}
}

function initHydrophoneDemo(){
  const demo=document.getElementById("hydrophone-demo"),svg=document.getElementById("hydrophone-svg"),source=document.getElementById("source-node"),waves=document.getElementById("wavefronts"),line=document.getElementById("estimate-line"),angle=document.getElementById("angle-readout"),dt=document.getElementById("dt-readout");
  if(!demo||!svg||!source||!waves||!line)return;
  const move=(clientX,clientY)=>{const rect=svg.getBoundingClientRect();const sx=Math.min(820,Math.max(100,(clientX-rect.left)/rect.width*1000));const sy=Math.min(310,Math.max(90,(clientY-rect.top)/rect.height*520));const tx=sx-225,ty=sy-145;source.setAttribute("transform",`translate(${tx} ${ty})`);waves.setAttribute("transform",`translate(${tx} ${ty})`);line.setAttribute("d",`M500 390L${sx.toFixed(1)} ${sy.toFixed(1)}`);const deg=Math.atan2(500-sx,390-sy)*180/Math.PI-90;if(angle)angle.value=`${Math.round(deg)}°`;if(dt){const d1=Math.hypot(sx-390,sy-390),d3=Math.hypot(sx-610,sy-390);const rel=(d3-d1)/220;dt.value=`REL ${rel>=0?"+":""}${rel.toFixed(2)}` }};
  demo.addEventListener("pointerdown",e=>move(e.clientX,e.clientY));if(isFinePointer)demo.addEventListener("pointermove",e=>move(e.clientX,e.clientY));
}

function initSkillGraph(){
  const details=[...document.querySelectorAll(".skill-groups details")],nodes=[...document.querySelectorAll(".skill-node")];
  details.forEach((d,i)=>d.addEventListener("toggle",()=>{if(d.open){nodes.forEach((n,idx)=>n.classList.toggle("is-active",idx===Math.min(i,nodes.length-1)))}}));
}

function initMagneticButtons(){
  if(!isFinePointer||prefersReducedMotion)return;
  document.querySelectorAll(".magnetic").forEach(el=>{el.addEventListener("pointermove",e=>{const r=el.getBoundingClientRect();const x=e.clientX-r.left-r.width/2,y=e.clientY-r.top-r.height/2;el.style.transform=`translate(${x*.08}px,${y*.12}px)`});el.addEventListener("pointerleave",()=>el.style.transform="")});
}

function initResumePlaceholder(){
  const toast=document.getElementById("resume-toast");document.querySelectorAll("[data-resume-trigger]").forEach(b=>b.addEventListener("click",()=>{toast?.classList.add("is-visible");window.clearTimeout(window.__resumeTimer);window.__resumeTimer=window.setTimeout(()=>toast?.classList.remove("is-visible"),4500)}));
}

function initDebugEasterEgg(){
  let buffer="";const out=document.getElementById("debug-output");
  window.addEventListener("keydown",e=>{if(e.key.length===1&&!e.metaKey&&!e.ctrlKey&&!e.altKey){buffer=(buffer+e.key.toLowerCase()).slice(-5);if(buffer==="debug")document.body.classList.toggle("debug-mode")}});
  if(isFinePointer)window.addEventListener("pointermove",e=>{if(document.body.classList.contains("debug-mode")&&out)out.value=`VIEWPORT / ${window.innerWidth}×${window.innerHeight} · CURSOR / ${e.clientX},${e.clientY}`},{passive:true});
}

document.addEventListener("DOMContentLoaded",()=>{initNavigation();initRevealAnimations();initScrollProgress();initProjectFilters();initPCBAnimation();initHydrophoneDemo();initSkillGraph();initMagneticButtons();initResumePlaceholder();initDebugEasterEgg()});
